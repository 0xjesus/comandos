pub fn verify()->Vec<serde_json::Value>{
 let mut rasters=Vec::new();
 let window=crate::Window::default();let other=crate::Window::default();let header=Header::new(&window,false);
 assert_eq!(header.controls.iter().map(|(_,n)|*n).collect::<Vec<_>>(),vec!["close","maximize","minimize"]);
 header.controls[1].0.click();header.controls[1].0.click();header.controls[2].0.click();header.controls[0].0.click();assert_eq!(*window.0.events.borrow(),vec!["maximize","unmaximize","iconify","close"]);assert!(other.0.events.borrow().is_empty());
 for (button,name) in &header.controls {let classes=button.0.2.context.0.borrow();assert!(classes.iter().any(|c|c=="cc-winctl"));assert_eq!(classes.iter().any(|c|c=="cc-close-btn"),*name=="close");}
 let themes=comandos_app::theme::themes_from_file(None);
 for name in ["noche","dia","calido","termius","bruno","superglass","neon","contraste","ubuntu"]{let theme=comandos_app::theme::desktop_theme(name,&themes).unwrap();header.paint(&theme);let brand=theme.values.get("brand").unwrap().as_str().unwrap();assert!(header.brand.0.0.borrow().contains(brand));for (b,_) in &header.controls{assert_eq!(b.0.2.image.borrow().as_ref().unwrap().pb.as_ref().unwrap().width(),16);}}
 for name in ["noche","dia","calido","termius","bruno","superglass","neon","contraste","ubuntu"]{
  let theme=comandos_app::theme::desktop_theme(name,&themes).unwrap();let color=theme.values.get("dim").unwrap().as_str().unwrap();
  for icon in ["plus","terminal","arrow-up-down","rows","chevron-left","chevron-right","panel-left","close","minimize","maximize","timer","bell","settings","sparkles"]{for size in [12,16,18,20]{let image=crate::icons::image(icon,size,color);let pb=image.pb.unwrap();rasters.push(serde_json::json!([name,icon,size,pb.width(),pb.height(),pb.read_pixel_bytes().as_ref()]));}}
 }
 let weak=window.downgrade();let button=header.controls[0].0.clone();drop(header);drop(window);assert!(weak.upgrade().is_none());button.click();
 rasters
}
